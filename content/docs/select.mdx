```svelte

  import { Select, SelectTrigger, SelectContent, SelectItem } from '$lib/components/select';
```

```svelte
    { value: 'admin', label: 'Admin' },
```

### Group

Use the `SelectGroup` component inside `SelectContent` to group related options together.

<DocsTabs items={["Preview", "Code"]}>
  <DocsTab value="Preview" className="mt-6">
    <Svelte url="/select/group" height={480} />
  </DocsTab>
  <DocsTab value="Code">

```svelte
<script lang="ts">
  import { Select, SelectTrigger, SelectContent, SelectGroup, SelectHeader, SelectItem, SelectSeparator } from '$lib/components/select';

  const roles = {
    management: [
      { value: 'admin', label: 'Admin', description: 'Full system access' },
      { value: 'editor', label: 'Editor', description: 'Can edit content' }
    ],
    limited: [
      { value: 'reviewer', label: 'Reviewer', description: 'Can review content' },
      { value: 'guest', label: 'Guest', description: 'Limited access' }
    ]
  };
</script>

<Select>
  <SelectTrigger placeholder="Assign user role" />
  <SelectContent>
    <SelectGroup>
      <SelectHeader>Management</SelectHeader>
      {#each roles.management as role}
        <SelectItem value={role.value} label={role.label}>
          {role.label}
        </SelectItem>
      {/each}
    </SelectGroup>
    <SelectSeparator />
    <SelectGroup>
      <SelectHeader>Limited access</SelectHeader>
      {#each roles.limited as role}
        <SelectItem value={role.value} label={role.label}>
          {role.label}
        </SelectItem>
      {/each}
    </SelectGroup>
  </SelectContent>
</Select>
```

  </DocsTab>
