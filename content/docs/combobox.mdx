import { SvelteComponent } from "@/components/SvelteComponent";
import { TypeTable } from "fumadocs-ui/components/type-table";

<SvelteComponent url="/combobox" />

```svelte
<script>
  import { Combobox, ComboboxInput, ComboboxContent, ComboboxItem } from '$lib/components/combobox';

  const fruits = [
    { value: 'apple', label: 'Apple' },
    { value: 'banana', label: 'Banana' },
    { value: 'cherry', label: 'Cherry' },
  ];
</script>

<Combobox name="fruit-selector" items={fruits}>
  <ComboboxInput />
  <ComboboxContent>
    {#each fruits as fruit (fruit.value)}
      <ComboboxItem value={fruit.value}>{fruit.label}</ComboboxItem>
    {/each}
  </ComboboxContent>
</Combobox>
```

## Filtering

The component supports filtering of items based on user input.

```svelte
<script>
  let inputValue = '';
  let touchedInput = false;

  $: filteredFruits = inputValue && touchedInput
    ? fruits.filter((fruit) => fruit.value.includes(inputValue.toLowerCase()))
    : fruits;
</script>

<Combobox name="combobox" items={filteredFruits}>
  <ComboboxInput bind:value={inputValue} />
  <ComboboxContent>
    {#each filteredFruits as fruit (fruit.value)}
      <ComboboxItem value={fruit.value} label={fruit.label}>
        {fruit.label}
      </ComboboxItem>
    {:else}
      <span>No results found</span>
    {/each}
  </ComboboxContent>
</Combobox>
```

## Custom Header

Add a header to the dropdown content using ComboboxHeader.

```svelte
<ComboboxContent>
  <ComboboxHeader>Select a Fruit</ComboboxHeader>
  <!-- ComboboxItems -->
</ComboboxContent>
```

## Item Indicator

Use ComboboxItemIndicator to add an icon or other indicator to selected items.

```svelte
<ComboboxItem value={fruit.value} label={fruit.label}>
  {fruit.label}
  <ComboboxItemIndicator>
    <Account />
  </ComboboxItemIndicator>
</ComboboxItem>
```

## Custom Arrow

Add a custom arrow to the combobox using ComboboxArrow.

```svelte
<Combobox>
  <!-- Other Combobox components -->
  <ComboboxArrow />
</Combobox>
```
