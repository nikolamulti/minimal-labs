## Usage

```svelte
<script lang="ts">
  import { Tooltip, TooltipTrigger, TooltipContent } from '$lib/components/tooltip';
  import { Button } from '$lib/components/button';
</script>
```

```svelte
<Tooltip>
  <TooltipTrigger>
    <Button>Hover me</Button>
  </TooltipTrigger>
  <TooltipContent>A simple tooltip that appears on hover</TooltipContent>
</Tooltip>
```

### Width

Use the `width` property to change the width of the tooltip content.

<DocsTabs items={["Preview", "Code"]}>
  <DocsTab value="Preview" className="mt-6">
    <Svelte url="/tooltip/width" height={360} />
  </DocsTab>
  <DocsTab value="Code">

```svelte
<script lang="ts">
  import { Tooltip, TooltipTrigger, TooltipContent } from '$lib/components/tooltip';
  import { Button } from '$lib/components/button';
</script>

<Tooltip>
  <TooltipTrigger>
    <Button>Hover me</Button>
  </TooltipTrigger>
  <TooltipContent width={320}>
    A simple tooltip that appears on hover. This is a tooltip with a fixed width of 320 pixels.
    The content will wrap to fit within this width constraint.
  </TooltipContent>
</Tooltip>
```
