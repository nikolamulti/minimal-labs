### Indeterminate

Use the `indeterminate` state to indicate that some, but not all, of the checkboxes in a group are selected.

<DocsTabs items={["Preview", "Code"]}>
  <DocsTab value="Preview" className="mt-6">
    <Svelte url="/checkbox/indeterminate" height={360} />
  </DocsTab>
  <DocsTab value="Code">

```svelte
<script lang="ts">
  import { Checkbox, CheckboxField, CheckboxGroup } from '$lib/components/checkbox';
  import { Label } from '$lib/components/fieldset';

  let options = [
    { id: 'option-1', label: 'Option 1', checked: true },
    { id: 'option-2', label: 'Option 2', checked: false },
    { id: 'option-3', label: 'Option 3', checked: false }
  ];

  $: allSelected = options.every((option) => option.checked);
  $: someSelected = options.some((option) => option.checked) && !allSelected;

  function handleSelectAll() {
    if (someSelected) {
      options = options.map((option) => ({ ...option, checked: true }));
    } else if (allSelected) {
      options = options.map((option) => ({ ...option, checked: false }));
    } else {
      options = options.map((option) => ({ ...option, checked: true }));
    }
  }
</script>

<CheckboxGroup>
  <CheckboxField>
    <Checkbox
      id="select-all"
      checked={someSelected ? 'indeterminate' : allSelected}
      on:click={handleSelectAll}
    />
    <Label for="select-all">Select all</Label>
  </CheckboxField>

  {#each options as option}
    <CheckboxField>
      <Checkbox id={option.id} bind:checked={option.checked} />
      <Label for={option.id}>{option.label}</Label>
    </CheckboxField>
  {/each}
</CheckboxGroup>
```

  </DocsTab>
</DocsTabs>
