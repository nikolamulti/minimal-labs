## Examples

### Icon support

Use the `icon` property to add an icon to the `TabsTrigger` component.

<DocsTabs items={["Preview", "Code"]}>
  <DocsTab value="Preview" className="mt-6">
    <Svelte url="/tabs/icon-support" height={360} />
  </DocsTab>
  <DocsTab value="Code">

```svelte
<script lang="ts">
  import { Tabs, TabsList, TabsTrigger, TabsContent } from '$lib/components/tabs';
  import { Menu, Settings, Account } from '$lib/components/icons';
</script>

<Tabs value="menu">
  <TabsList>
    <TabsTrigger icon={Menu} value="menu">Menu</TabsTrigger>
    <TabsTrigger icon={Settings} value="settings">Settings</TabsTrigger>
    <TabsTrigger icon={Account} value="account">Account</TabsTrigger>
  </TabsList>
  <TabsContent value="menu"></TabsContent>
  <TabsContent value="settings"></TabsContent>
  <TabsContent value="account"></TabsContent>
</Tabs>
```

  </DocsTab>
</DocsTabs>
