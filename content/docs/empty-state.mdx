```svelte
  import { EmptyState, EmptyStateTitle, EmptyStateDescription, EmptyStateAction } from '@/components/empty-state';
  import { Button } from '@/components/button';
  import { Add } from '@/components/icons';
</script>

<EmptyState icon={Add}>
  <EmptyStateTitle>No links found</EmptyStateTitle>

```

## Component API

The EmptyState component and its subcomponents accept the following props:

  type={{

>

## Variants

The EmptyState component supports two main variants:

<SvelteComponent height={160} url="/empty-state/variants" />

```svelte
<EmptyState icon={Add}>
		<EmptyStateTitle>No links found</EmptyStateTitle>
		<EmptyStateDescription>Create a new link to get started.</EmptyStateDescription>
		<EmptyStateAction>
			<Link variant="primary" href="/links/create">Link text</Link>
		</EmptyStateAction>
	</EmptyState>
	<EmptyState transparent icon={Add}>
		<EmptyStateTitle>No links found</EmptyStateTitle>
		<EmptyStateDescription>Create a new link to get started.</EmptyStateDescription>
		<EmptyStateAction>
			<Link variant="primary" href="/links/create">Link text</Link>
		</EmptyStateAction>
	</EmptyState>
```
