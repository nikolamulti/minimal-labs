### Checkbox

Use the `DropdownCheckboxItem` component to create a checkbox item.

<DocsTabs items={["Preview", "Code"]}>
  <DocsTab value="Preview" className="mt-6">
    <Svelte url="/dropdown-menu/checkbox" height={400} />
  </DocsTab>
  <DocsTab value="Code">

```svelte
<script lang="ts">
  import { Dropdown, DropdownTrigger, DropdownContent, DropdownCheckboxItem } from '$lib/components/dropdown-menu';
  import { ChevronSmallDown } from '$lib/components/icons';
  import { Button } from '$lib/components/button';

  const account = {
    personalAccount: false,
    businessAccount: false,
    enterpriseAccount: false
  };
</script>

<Dropdown closeOnItemClick={false}>
  <DropdownTrigger asChild let:builder>
    <Button builders={[builder]} trailingIcon={ChevronSmallDown}>Account</Button>
  </DropdownTrigger>
  <DropdownContent class="dropdown__content">
    <DropdownCheckboxItem bind:checked={account.personalAccount}>
      Personal account
    </DropdownCheckboxItem>
    <DropdownCheckboxItem bind:checked={account.businessAccount}>
      Business account
    </DropdownCheckboxItem>
    <DropdownCheckboxItem bind:checked={account.enterpriseAccount}>
      Enterprise account
    </DropdownCheckboxItem>
  </DropdownContent>
</Dropdown>

<style>
  :global(.dropdown__content) {
    width: 200px;
  }
```
